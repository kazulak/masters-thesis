cmake_minimum_required(VERSION 3.21)
project(thesis_evaluation_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED QUEST_SOURCE OR NOT DEFINED EVAL_PRECISION)
  message(FATAL_ERROR "Supply QUEST_SOURCE and EVAL_PRECISION=1 or 2")
endif()
if(NOT EVAL_PRECISION MATCHES "^[12]$")
  message(FATAL_ERROR "EVAL_PRECISION must be 1 or 2")
endif()
set(FLOAT_PRECISION ${EVAL_PRECISION} CACHE STRING "" FORCE)
set(ENABLE_MULTITHREADING ON CACHE BOOL "" FORCE)
set(ENABLE_DISTRIBUTION OFF CACHE BOOL "" FORCE)
set(ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(ENABLE_HIP OFF CACHE BOOL "" FORCE)
set(ENABLE_CUQUANTUM OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${QUEST_SOURCE} quest-build)
find_package(OpenMP REQUIRED)
add_library(quest_bridge SHARED quest_bridge.cpp)
target_include_directories(quest_bridge PRIVATE ${QUEST_SOURCE} ${CMAKE_BINARY_DIR}/quest-build)
target_compile_options(quest_bridge PRIVATE -O3 -march=native -Wall -Wextra)
target_link_libraries(quest_bridge PRIVATE QuEST OpenMP::OpenMP_CXX)
set_target_properties(quest_bridge PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
